function toHex(buffer) {

function constantTimeEqual(a, b) {
  if (a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i += 1) {
    diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }
  return diff === 0;
}
async function hmacSha256(keyBytes, message) {
  const key = await crypto.subtle.importKey(
    "raw",
    keyBytes,
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"]
  );
  return crypto.subtle.sign("HMAC", key, encoder.encode(message));
}
function buildDataCheckString(entries) {
  return Array.from(entries).filter(([key]) => key !== "hash" && key !== "signature").sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0).map(([key, value]) => `${key}=${value}`).join("\n");
}
async function signInitData(initData, botToken) {
  const params = new URLSearchParams(initData);
  params.delete("hash");
  params.delete("signature");
  const dataCheckString = buildDataCheckString(params.entries());
  const secret = await hmacSha256(encoder.encode("WebAppData"), botToken);
  const signature = await hmacSha256(secret, dataCheckString);
  return toHex(signature);
}
async function verifyInitData(rawInitData, botToken, options = {}) {

      "missing_init_data",
      "Open Taskon from inside Telegram so we can verify you."
    );
  }
  const nowSeconds = Math.floor((options.now ?? Date.now()) / 1e3);
  const maxAge = options.maxAgeSeconds ?? 60 * 60 * 24;

    throw new TelegramAuthError("server_misconfigured", "Bot token is missing on the server.");
  }
  const params = new URLSearchParams(rawInitData);
  const hash = params.get("hash");
  if (!hash) {
    throw new TelegramAuthError("invalid_init_data", "This session is missing its signature.");
  }
  const authDateRaw = params.get("auth_date");
  const authDate = Number(authDateRaw);
  if (!authDateRaw || !Number.isFinite(authDate)) {
    throw new TelegramAuthError("invalid_init_data", "This session has no timestamp.");
  }
  if (nowSeconds - authDate > maxAge) {
    throw new TelegramAuthError(
      "init_data_expired",
      "This session is old. Close Taskon and open it again."
    );
  }
  if (authDate - nowSeconds > 120) {
    throw new TelegramAuthError("invalid_init_data", "This session timestamp is in the future.");
  }
  const expected = await signInitData(rawInitData, botToken);
  if (!constantTimeEqual(expected, hash.toLowerCase())) {
    throw new TelegramAuthError(
      "invalid_init_data",
      "We couldn't verify this Telegram session. Reopen the app from the bot."
    );
  }
  let user;
  try {
    user = JSON.parse(params.get("user") ?? "");
  } catch {
    throw new TelegramAuthError("invalid_init_data", "This session has no user.");
  }
  if (!user || !Number.isSafeInteger(user.id) || user.id <= 0) {
    throw new TelegramAuthError("invalid_init_data", "This session has an invalid user.");
  }
  return {
    user,
    startParam: params.get("start_param"),
    authDate
  };
}

async function buildSession(db, ctx) {

  const yesterday = isoDate(new Date(now.getTime() - 864e5));

  const projectedStreak = doneToday ? ctx.user.streak + 1 : currentStreak + 1;
  const bonus = bonusForStreak(projectedStreak);
  const reward = BASE_CHECKIN + bonus;
  const [
    tasksResult,
    completedResult,

    recentReferralsResult,
    rankResult,
    totalResult

    db.from("tasks").select("id,title,type,reward,active,expires_at,created_at").order("created_at"),
    db.from("user_tasks").select("task_id").eq("user_id", ctx.user.id),
    db.from("referrals").select("id", { count: "exact", head: true }).eq("referrer_id", ctx.user.id),
    db.from("referrals").select("referrer_bonus").eq("referrer_id", ctx.user.id).eq("bonus_paid", true),

  ]);

  if (error) throw fromDbError(error);
  const row = Array.isArray(data) ? data[0] : null;
  if (!row) throw fromDbError("user_not_found");
  return {
    points: Number(row.new_points),
    streak: Number(row.new_streak),
    reward: Number(row.reward),
    bonus: Number(row.bonus)
  };
}

// supabase/functions/_shared/links.ts
function channelRefFromUrl(rawUrl) {
  const value = rawUrl.trim();
  if (value.startsWith("@")) return value;
  if (/^-?\d{5,}$/.test(value)) return value;
  try {
    const url = new URL(value);
    if (!["t.me", "telegram.me", "telegram.dog"].includes(url.hostname)) return null;
    const segment = url.pathname.split("/").filter(Boolean)[0];
    if (!segment || segment.startsWith("+") || segment.toLowerCase() === "joinchat") {
      return null;
    }
    return `@${segment}`;
  } catch {
    return null;
  }
}

// supabase/functions/_shared/tasks.ts
async function listTasks(db, userId) {
  const [tasksResult, completedResult, startsResult] = await Promise.all([
    db.from("tasks").select("*").order("created_at", { ascending: true }),
    db.from("user_tasks").select("task_id,completed_at").eq("user_id", userId),
    db.from("user_task_starts").select("task_id,started_at").eq("user_id", userId)
  ]);
  if (tasksResult.error) throw fromDbError(tasksResult.error);
  const completed = new Map(
    (completedResult.data ?? []).map((row) => [row.task_id, row.completed_at])
  );
  const starts = new Map(
    (startsResult.data ?? []).map((row) => [row.task_id, row.started_at])
  );
  const now = Date.now();
  const views = [];
  for (const task of tasksResult.data ?? []) {
    const completedAt = completed.get(task.id);
    const startedAt = starts.get(task.id);
    const expired = task.expires_at ? new Date(task.expires_at).getTime() <= now : false;
    if (!completedAt) {
      if (!task.active || expired) continue;
    }
    const timerSeconds = task.type === "join_channel" ? 0 : task.timer_seconds;
    const view = {
      id: task.id,
      title: task.title,
      type: task.type,
      reward: task.reward,
      url: task.url,
      timerSeconds,
      expiresAt: task.expires_at,
      status: completedAt ? "completed" : startedAt ? "started" : "available"
    };
    if (startedAt) {
      view.startedAt = startedAt;
      view.claimAt = new Date(new Date(startedAt).getTime() + timerSeconds * 1e3).toISOString();
    }
    if (completedAt) view.completedAt = completedAt;
    views.push(view);
  }
  return views.sort((a, b) => {
    const rank = { available: 0, started: 1, completed: 2 };
    if (rank[a.status] !== rank[b.status]) return rank[a.status] - rank[b.status];
    return 0;
  });
}
async function startTask(db, userId, taskId) {
  const { data, error } = await db.rpc("start_task", {
    p_user_id: userId,
    p_task_id: taskId
  });
  if (error) throw fromDbError(error);
  const row = Array.isArray(data) ? data[0] : null;
  if (!row) throw fromDbError("task_unavailable");
  return {
    startedAt: row.started_at,
    timerSeconds: Number(row.timer_seconds),
    claimAt: row.claim_at
  };
}
async function isChannelMember(botToken, taskUrl, telegramId) {
  const chatId = channelRefFromUrl(taskUrl);
  if (!chatId) {
    throw new ApiError(
      "verification_failed",
      "This channel link can't be checked. Ask the admin to use a public @channel link.",
      400
    );
  }
  const response = await fetch(`https://api.telegram.org/bot${botToken}/getChatMember`, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ chat_id: chatId, user_id: telegramId })
  });
  const body = await response.json().catch(() => null);
  if (!response.ok || !body?.ok) {
    console.error("getChatMember failed:", body?.description ?? response.status);
    throw new ApiError(
      "verification_failed",
      "We can't check that channel right now. The bot may need to be added to it.",
      400
    );
  }
  const status = body.result?.status;
  return status === "creator" || status === "administrator" || status === "member" || status === "restricted" && body.result?.is_member === true;
}
async function claimTask(db, user, taskId, botToken) {
  const { data: task, error } = await db.from("tasks").select("id,type,url,active,expires_at").eq("id", taskId).maybeSingle();
  if (error) throw fromDbError(error);
  if (!task) throw fromDbError("task_unavailable");
  let verified = false;
  if (task.type === "join_channel") {
    if (!botToken) {
      throw new ApiError("server_misconfigured", MESSAGES.server_misconfigured, 500);
    }
    verified = await isChannelMember(botToken, task.url, Number(user.telegram_id));
    if (!verified) {
      throw new ApiError("membership_not_found", MESSAGES.membership_not_found, 403);
    }
  }
  const { data, error: claimError } = await db.rpc("complete_task", {
    p_user_id: user.id,
    p_task_id: taskId,
    p_verified: verified
  });
  if (claimError) throw fromDbError(claimError);
  const row = Array.isArray(data) ? data[0] : null;
  if (!row) throw fromDbError("internal_error");
  return { points: Number(row.new_points), reward: Number(row.reward) };
}

// supabase/functions/api/index.ts
function payloadOf(body) {
  return typeof body.payload === "object" && body.payload !== null ? body.payload : {};
}
function requireTaskId(payload) {
  if (!isUuid(payload.taskId)) {
    throw new ApiError("task_unavailable", MESSAGES.task_unavailable, 400);
  }
  return payload.taskId;
}
Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return preflight(req);
  try {
    if (req.method !== "POST") {
      throw new ApiError("bad_request", MESSAGES.bad_request, 405);
    }
    const body = await req.json().catch(() => null);
    if (!body || typeof body !== "object") {
      throw new ApiError("bad_request", MESSAGES.bad_request, 400);
    }
    const action = typeof body.action === "string" ? body.action : "";
    if (!action) throw new ApiError("bad_request", MESSAGES.bad_request, 400);
    const db = getServiceClient();
    const ctx = await authenticate(req, db, body.initData);
    const payload = payloadOf(body);
    const isAdminAction = action.startsWith("admin.");
    await enforceRateLimit(db, ctx.user.id, "api", 180, 60);
    switch (action) {
      case "session": {
        await enforceRateLimit(db, ctx.user.id, "session", 90, 60);
        return json(req, { data: await buildSession(db, ctx) });
      }
      case "checkin": {
        await enforceRateLimit(db, ctx.user.id, "checkin", 10, 3600);
        const result = await claimCheckin(db, ctx.user.id);
        return json(req, { data: result });
      }
      case "tasks.list": {
        await enforceRateLimit(db, ctx.user.id, "tasks.list", 90, 60);
        return json(req, { data: await listTasks(db, ctx.user.id) });
      }
      case "tasks.start": {
        await enforceRateLimit(db, ctx.user.id, "tasks.start", 30, 60);
        const taskId = requireTaskId(payload);
        return json(req, { data: await startTask(db, ctx.user.id, taskId) });
      }
      case "tasks.claim": {
        await enforceRateLimit(db, ctx.user.id, "tasks.claim", 30, 60);
        const taskId = requireTaskId(payload);
        const result = await claimTask(
          db,
          ctx.user,
          taskId,
          Deno.env.get("BOT_TOKEN") ?? ""
        );
        return json(req, { data: result });
      }
      case "referrals": {
        await enforceRateLimit(db, ctx.user.id, "referrals", 60, 60);
        const session = await buildSession(db, ctx);
        return json(req, {
          data: {
            referral: session.referral,
            counts: session.counts
          }
        });
      }
      case "leaderboard": {
        await enforceRateLimit(db, ctx.user.id, "leaderboard", 60, 60);
        const [board, total] = await Promise.all([
          db.rpc("get_leaderboard", { p_user_id: ctx.user.id, p_limit: 100 }),
          db.from("users").select("id", { count: "exact", head: true }).eq("banned", false)
        ]);
        if (board.error) throw fromDbError(board.error);
        const rows = (board.data ?? []).map((row) => ({
          position: Number(row.place),
          telegramId: row.telegram_id,
          username: row.username,
          firstName: row.first_name,
          points: row.points,
          isMe: row.is_me === true
        }));
        return json(req, {
          data: {
            top: rows.filter((row) => row.position <= 100),
            me: rows.find((row) => row.isMe) ?? null,
            total: total.count ?? 0
          }
        });
      }
      default: {
        if (isAdminAction) {
          return json(req, {
            data: await handleAdmin(db, ctx, action, payload)
          });
        }
        throw new ApiError("bad_request", "Unknown action.", 404);
      }
    }
  } catch (error) {
    if (error instanceof ApiError) {
      return json(
        req,
        { error: { code: error.code, message: error.message } },
        error.status
      );
    }
    console.error("Unhandled API error:", error);
    return json(
      req,
      { error: { code: "internal_error", message: MESSAGES.internal_error } },
      500
    );
  }
});
